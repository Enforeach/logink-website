// Budget: one preloaded file. Archivo latin woff2, wdth 62–125 × wght 100–900 = 90,096 B
// (fvar default wght 600). Martian Mono 400 = 10,376 B, lazy. Total 100,472 B.
import { Archivo, Martian_Mono } from 'next/font/google'

export const archivo = Archivo({
  subsets: ['latin'],
  axes: ['wdth'], // weight omitted = variable wght; CSS uses font-stretch 62%–125% (maps to wdth)
  display: 'swap',
  preload: true,
  adjustFontFallback: false, // replaced by the measured 'LK *' faces in tokens.css (one per cut)
  variable: '--font-archivo', // = "'__Archivo_<hash>'" only; stacks live in tokens.css (--ff-*)
})

export const martian = Martian_Mono({
  subsets: ['latin'],
  weight: '400',
  display: 'swap',
  preload: false, // instruments + captions only
  adjustFontFallback: false, // 'LK Mono C' / 'LK Mono A' in tokens.css
  variable: '--font-martian',
})
