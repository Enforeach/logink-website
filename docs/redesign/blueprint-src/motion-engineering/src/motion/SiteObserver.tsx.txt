'use client'

// T1: the only motion JS every device downloads (budget <= 1.5 KB gz). Mount once per root layout
// (id and en), outside <main>. Server Components opt in with data attributes; nothing imports motion.
//
// Contract
//   html.js            set HERE, after the first scan (not by an inline head script): if this chunk is
//                      blocked or slow, nothing is ever hidden. CSS hides pre-reveal states only under
//                      .js AND (prefers-reduced-motion: no-preference).
//   [data-reveal]      -> data-in once it enters (rootMargin -8% bottom). Elements already on screen at
//                      scan time get data-in BEFORE .js exists, so they never animate or flash. In a
//                      12,140 px render viewport everything is on screen, so nothing is ever masked.
//                      Values: lift | fade (decorative only, never text) | split (H2 only) | words | wipe | inst.
//   [data-steps]       -> data-step="k" on the section, data-on on its [data-step-panel][data-k=k],
//                      aria-current="step" on the k-th [data-step-item] (item crossing 58-60% of the viewport).
//   [data-sky]         -> html[data-sky] = light|dark of the deepest [data-sky] under the header band (top 8%).
//                      Every Section emits data-sky from its ground; <main data-sky> seeds the pre-hydration
//                      header (design system: :root:not([data-sky]):has(main[data-sky=dark])).
//   html[data-nav]     top | pill (> 80 px) | hide (scrolling down past 240 px). No rAF, no setState.
//   #mode-teks         checkbox (footer; Bab III uses <label for="mode-teks">): html.mode-teks + localStorage
//                      'lk-teks' + a window 'lk:teks' event that the JS gates listen to.
//   [commandfor]       <button commandfor="menu" command="show-modal"> opens the menu <dialog> natively
//                      (Invoker Commands); older browsers get the same behaviour from the click fallback.
import { useLayoutEffect, useSyncExternalStore } from 'react'
import { usePathname } from 'next/navigation'

export function SiteObserver() {
  const path = usePathname()

  // Reveal + steps + sky: rescanned on every route change (layouts persist across navigations).
  // Layout effect: runs before the new page paints, so on-screen elements are marked in time.
  useLayoutEffect(() => {
    const html = document.documentElement
    const vh = innerHeight

    const reveal = new IntersectionObserver(
      (es) => {
        for (const e of es) {
          if (!e.isIntersecting) continue
          if (e.target.getAttribute('data-reveal') === 'split') indexLines(e.target)
          e.target.setAttribute('data-in', '')
          reveal.unobserve(e.target)
        }
      },
      { rootMargin: '0px 0px -8% 0px' },
    )
    // Read every rect first, then write: one layout, not N.
    const els = [...document.querySelectorAll('[data-reveal]:not([data-in])')]
    const onScreen = els.map((el) => {
      const r = el.getBoundingClientRect()
      return r.top < vh && r.bottom > 0
    })
    els.forEach((el, i) => (onScreen[i] ? el.setAttribute('data-in', '') : reveal.observe(el)))
    html.classList.add('js')

    const steps = new IntersectionObserver(
      (es) => {
        for (const e of es) {
          if (!e.isIntersecting) continue
          const sec = e.target.closest('[data-steps]')
          if (!sec) continue
          const items = [...sec.querySelectorAll('[data-step-item]')]
          const k = String(items.indexOf(e.target))
          if (sec.getAttribute('data-step') === k) continue
          sec.setAttribute('data-step', k)
          items.forEach((it, i) => (String(i) === k ? it.setAttribute('aria-current', 'step') : it.removeAttribute('aria-current')))
          sec.querySelectorAll('[data-step-panel]').forEach((p) => p.toggleAttribute('data-on', p.getAttribute('data-k') === k))
        }
      },
      // A 2% band at 58-60% of the viewport: mid reading area on desktop, just under the 40svh sticky
      // band on phones. One item at a time; the latest crossing wins.
      { rootMargin: '-58% 0px -40% 0px' },
    )
    document.querySelectorAll('[data-steps] [data-step-item]').forEach((el) => steps.observe(el))

    // Sky: keep the set of [data-sky] elements under the header band and pick the deepest (last in
    // document order), so a nested band (e.g. the Malam closing band inside a Kertas page) wins while
    // it is there and the outer section takes over again when it leaves.
    const live = new Set<Element>()
    const sky = new IntersectionObserver(
      (es) => {
        for (const e of es) {
          if (e.isIntersecting) live.add(e.target)
          else live.delete(e.target)
        }
        let pick: Element | undefined
        for (const el of live) if (!pick || pick.compareDocumentPosition(el) & Node.DOCUMENT_POSITION_FOLLOWING) pick = el
        const v = pick?.getAttribute('data-sky') || 'light'
        if (html.getAttribute('data-sky') !== v) html.setAttribute('data-sky', v)
      },
      { rootMargin: '0px 0px -92% 0px' }, // the band under the header
    )
    document.querySelectorAll('[data-sky]').forEach((el) => sky.observe(el))

    return () => {
      reveal.disconnect()
      steps.disconnect()
      sky.disconnect()
    }
  }, [path])

  // Header state, Mode teks, dialog fallback: page-independent, bound once.
  useLayoutEffect(() => {
    const html = document.documentElement
    let last = scrollY
    const nav = () => {
      const y = scrollY // reading scrollY forces no layout
      if (y >= 80 && Math.abs(y - last) < 4) return // ignore 1-3 px trackpad jitter
      const v = y < 80 ? 'top' : y > last && y > 240 ? 'hide' : 'pill'
      last = y
      if (html.dataset.nav !== v) html.dataset.nav = v
    }
    nav()
    addEventListener('scroll', nav, { passive: true })

    const box = document.getElementById('mode-teks') as HTMLInputElement | null
    if (box) box.checked = html.classList.contains('mode-teks')
    const change = (e: Event) => {
      const t = e.target as HTMLInputElement
      if (t.id !== 'mode-teks') return
      html.classList.toggle('mode-teks', t.checked)
      try {
        if (t.checked) localStorage.setItem('lk-teks', '1')
        else localStorage.removeItem('lk-teks')
      } catch {}
      dispatchEvent(new Event('lk:teks'))
    }
    document.addEventListener('change', change)

    // Invoker Commands fallback (Safari < 26.2, Firefox < 144): only when the platform lacks it.
    const native = 'commandForElement' in HTMLButtonElement.prototype
    const click = (e: MouseEvent) => {
      const b = (e.target as Element).closest?.('button[commandfor]')
      const d = b && document.getElementById(b.getAttribute('commandfor')!)
      if (!(d instanceof HTMLDialogElement)) return
      const cmd = b!.getAttribute('command')
      if (cmd === 'show-modal' && !d.open) d.showModal()
      else if (cmd === 'close') d.close()
    }
    if (!native) document.addEventListener('click', click)

    return () => {
      removeEventListener('scroll', nav)
      document.removeEventListener('change', change)
      document.removeEventListener('click', click)
    }
  }, [])

  return null
}

// Per-line stagger for data-reveal="split": word spans get --line from their offsetTop at reveal
// time (fonts are loaded by then). Two passes: read all, then write.
function indexLines(el: Element) {
  const words = [...el.querySelectorAll<HTMLElement>('.lk-w')]
  const tops = words.map((w) => w.offsetTop)
  let line = -1
  let top = -Infinity
  words.forEach((w, i) => {
    if (tops[i] > top + 2) {
      line++
      top = tops[i]
    }
    w.style.setProperty('--line', String(line))
  })
}

/** For the header island: re-render only when html[data-sky] changes (ONE LogoFull, theme prop). */
export function useHtmlAttr(name: string): string | null {
  return useSyncExternalStore(
    (cb) => {
      const mo = new MutationObserver(cb)
      mo.observe(document.documentElement, { attributes: true, attributeFilter: [name] })
      return () => mo.disconnect()
    },
    () => document.documentElement.getAttribute(name),
    () => null,
  )
}
