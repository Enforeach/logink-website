// No-JS, reduced-motion and Mode teks states: BASE=... H1_JSON=h1.json node states.cjs [/path ...]
// Screens go to acceptance-out/*.png. Asserts: approved H1 string, 0 hidden text in every viewport the
// page scrolls through (RM, no-JS), no inline opacity:0 in the SSR HTML, JSON-LD strings visible,
// no rich/player chunk under reduced motion or Mode teks, no img/video/canvas painted in Mode teks.
const fs = require('fs')
const path = require('path')
const { chromium, PHONE, DESKTOP, HEADERS, recordScripts, markersIn, hiddenTextInPage, jsonLdVisibleInPage, report } = require('./lib.cjs')
const BASE = process.env.BASE || 'http://localhost:3000'
const PATHS = process.argv.slice(2).length ? process.argv.slice(2) : ['/', '/layanan/jasa-seo-profesional']
const H1 = process.env.H1_JSON ? JSON.parse(fs.readFileSync(process.env.H1_JSON, 'utf8')) : {}
const OUT = path.join(process.cwd(), 'acceptance-out')
fs.mkdirSync(OUT, { recursive: true })
const slug = (p) => p.replace(/\W+/g, '_') || 'home'

async function sweep(page) {
  // Scroll through the page one viewport at a time and collect hidden text at each stop.
  const hidden = []
  const h = await page.evaluate(() => document.documentElement.scrollHeight)
  const vh = page.viewportSize().height
  for (let y = 0; y < h; y += Math.round(vh * 0.8)) {
    await page.evaluate((y) => scrollTo(0, y), y)
    await page.waitForTimeout(1300) // reveals (<= 900 ms) settle
    const r = await page.evaluate(hiddenTextInPage)
    hidden.push(...r.hidden.map((x) => ({ ...x, y })))
  }
  return hidden
}

;(async () => {
  const browser = await chromium.launch()
  const rows = []
  const failures = []
  for (const p of PATHS) {
    const html = await (await fetch(BASE + p, { headers: HEADERS })).text()
    const inlineHidden = html.match(/style="[^"]*opacity:\s*0(\.0+)?\s*[;"]/g) || []
    if (inlineHidden.length) failures.push(`${p} SSR has ${inlineHidden.length} inline opacity:0 styles`)
    for (const [name, opts] of [['nojs-390', { ...PHONE, viewport: { width: 390, height: 844 }, javaScriptEnabled: false }], ['nojs-1440', { ...DESKTOP, javaScriptEnabled: false }], ['rm-390', { ...PHONE, viewport: { width: 390, height: 844 }, reducedMotion: 'reduce' }], ['rm-1440', { ...DESKTOP, reducedMotion: 'reduce' }], ['teks-1440', { ...DESKTOP }]]) {
      const ctx = await browser.newContext(opts)
      if (name.startsWith('teks')) await ctx.addInitScript(() => localStorage.setItem('lk-teks', '1'))
      const page = await ctx.newPage()
      const scripts = recordScripts(page)
      await page.goto(BASE + p, { waitUntil: 'load' })
      await page.waitForTimeout(2500)
      const h1 = await page.evaluate(() => [...document.querySelectorAll('h1')].map((h) => h.textContent.replace(/\s+/g, ' ').trim()))
      if (h1.length !== 1) failures.push(`${p} ${name}: ${h1.length} h1 elements`)
      if (H1[p] && h1[0] !== H1[p]) failures.push(`${p} ${name}: h1 "${h1[0]}" !== approved "${H1[p]}"`)
      await page.screenshot({ path: path.join(OUT, `${slug(p)}-${name}.png`), fullPage: true })
      const row = { path: p, state: name, h1: h1[0] }
      if (!name.startsWith('teks')) {
        const hidden = await sweep(page)
        row.hidden = hidden.length
        if (hidden.length) failures.push(`${p} ${name}: ${hidden.length} hidden text, e.g. ${JSON.stringify(hidden.slice(0, 3))}`)
      } else {
        const painted = await page.evaluate(() => [...document.querySelectorAll('img,video,canvas')].filter((e) => e.getClientRects().length).length)
        row.painted = painted
        if (painted) failures.push(`${p} ${name}: ${painted} img/video/canvas still painted in Mode teks`)
      }
      if (name === 'nojs-1440') {
        const miss = await page.evaluate(jsonLdVisibleInPage)
        row.jsonLdMissing = miss.length
        if (miss.length) failures.push(`${p}: ${miss.length} JSON-LD strings not visible, e.g. ${miss[0]}`)
      }
      if (name.startsWith('rm') || name.startsWith('teks')) {
        const mk = markersIn(scripts)
        row.markers = mk
        for (const k of ['lenis', 'rich', 'player']) if (mk[k].length) failures.push(`${p} ${name}: loaded ${k} code (${mk[k].join(', ')})`)
      }
      rows.push(row)
      await ctx.close()
    }
  }
  await browser.close()
  report('states', rows, failures)
})()
