// Debug probe: which section grows with the viewport height (tall-viewport ratio failures).
// BASE=... node heights.cjs [/path]
const { chromium, PHONE } = require('./acceptance/lib.cjs')
const BASE = process.env.BASE || 'http://localhost:4180'
;(async () => {
  const b = await chromium.launch()
  for (const h of [732, 12140]) {
    const ctx = await b.newContext({ ...PHONE, viewport: { width: 412, height: h } })
    const p = await ctx.newPage()
    await p.goto(BASE + (process.argv[2] || '/'), { waitUntil: 'load' })
    await p.waitForTimeout(1500)
    const rows = await p.evaluate(() =>
      [...document.querySelectorAll('body *')]
        .filter((e) => e.getBoundingClientRect().height > 200)
        .map((e) => ({ el: `${e.tagName.toLowerCase()}.${String(e.className).split(' ')[0]}`, h: Math.round(e.getBoundingClientRect().height) })),
    )
    console.log(h, JSON.stringify(rows.slice(0, 40)))
    await ctx.close()
  }
  await b.close()
})()
